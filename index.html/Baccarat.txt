  <script>
    let resultados = [];

    function addResultado(resultado) {
      resultados.push(resultado);
      atualizar();
    }

    function atualizar() {
      let jogador = resultados.filter(r => r === 'Jogador').length;
      let banca = resultados.filter(r => r === 'Banca').length;
      let empate = resultados.filter(r => r === 'Empate').length;

      let total = jogador + banca;

      let chanceJogador = total > 0
        ? ((jogador / total) * 100).toFixed(1)
        : 0;

      let chanceBanca = total > 0
        ? ((banca / total) * 100).toFixed(1)
        : 0;

      document.getElementById('jogador').innerText = jogador;
      document.getElementById('banca').innerText = banca;
      document.getElementById('empate').innerText = empate;

      document.getElementById('chanceJogador').innerText = chanceJogador + '%';
      document.getElementById('chanceBanca').innerText = chanceBanca + '%';

      let sinal = 'Sem sinal';

      if (chanceJogador > chanceBanca) {
        sinal = 'Maior tendência: JOGADOR';
      }

      if (chanceBanca > chanceJogador) {
        sinal = 'Maior tendência: BANCA';
      }

      document.getElementById('sinal').innerText = sinal;

      document.getElementById('historico').innerHTML =
        '<h3>Histórico</h3>' + resultados.join(' | ');
    }
  </script>
